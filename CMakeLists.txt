cmake_minimum_required(VERSION 3.16)
project(AudioPlayer VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(QT_VERSION_MAJOR "" CACHE STRING "Force Qt major version (5 or 6); empty prefers Qt 6")
if(QT_VERSION_MAJOR)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Gui Widgets Quick QuickControls2 Concurrent)
else()
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Quick)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Gui Widgets Quick QuickControls2 Concurrent)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat>=59 libavcodec>=59 libavutil>=57 libswresample>=4)
pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
add_library(player_core STATIC
    audiodecoder.cpp audioplayer.cpp playbackworker.cpp playercontroller.cpp
    playlistmodel.cpp playliststore.cpp libraryimporter.cpp)
target_include_directories(player_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(player_core PUBLIC Qt${QT_VERSION_MAJOR}::Gui PkgConfig::FFMPEG PkgConfig::PORTAUDIO)
# Qt Widgets is only used for native file dialogs; the entire player UI is QML.
add_library(player_app STATIC librarycontroller.cpp)
target_link_libraries(player_app PUBLIC player_core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Concurrent)
add_executable(audio_player main.cpp resources.qrc)
target_link_libraries(audio_player PRIVATE player_app Qt${QT_VERSION_MAJOR}::Quick Qt${QT_VERSION_MAJOR}::QuickControls2)
foreach(target player_core player_app audio_player)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_subdirectory(tests)
endif()
install(TARGETS audio_player RUNTIME DESTINATION bin)
